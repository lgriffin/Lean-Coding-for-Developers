# deploy-checklist.yml
pre_deployment:
  - verify_config:
      check: "DB_HOST matches staging endpoint"
      how: "diff .env.staging against known-good config"
      automated: false   # TODO: add CI check
  - verify_migrations:
      check: "All pending migrations applied to staging DB"
      how: "run: python manage.py showmigrations --plan"
  - verify_tests:
      check: "All tests pass on this commit"
      how: "check CI status badge on the PR"
  - verify_secrets:
      check: "No hardcoded credentials in diff"
      how: "run: git diff HEAD~1 | grep -i 'password\|secret\|key'"
  - verify_rollback:
      check: "Previous deployment tag is recorded"
      how: "note current deployed commit SHA before deploying"
